cmake_minimum_required(VERSION 3.16)
project(dm_inventory_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FIC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${FIC_SOURCE}/cmake/FicTargetPlatform.cmake")
add_subdirectory("${FIC_SOURCE}/fic-common/fic-core" fic-core)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}/platform")
foreach(header PasswordAgingPolicyDefaultsGenerated SudoSecurePathDefaultGenerated UserCreationPolicyDefaultsGenerated)
    configure_file("${FIC_SOURCE}/fic/src/platform/generated/${header}.h.in"
        "${generated}/platform/${header}.h" @ONLY)
endforeach()
foreach(variant batch individual)
    set(inventory "${FIC_SOURCE}/fic/src/modules/oss/display_manager/DisplayManagerInventory.cpp")
    if(variant STREQUAL "individual")
        if(NOT DEFINED FIC_DM_BASELINE_SOURCE)
            message(FATAL_ERROR "FIC_DM_BASELINE_SOURCE must name the original inventory source")
        endif()
        set(inventory "${FIC_DM_BASELINE_SOURCE}")
    endif()
    add_executable(dm_inventory_${variant} Driver.cpp "${inventory}"
        "${FIC_SOURCE}/fic/src/platform/PlatformExecutableResolver.cpp")
    target_include_directories(dm_inventory_${variant} PRIVATE
        "${FIC_SOURCE}/fic/src" "${generated}")
    target_link_libraries(dm_inventory_${variant} PRIVATE fic-core)
endforeach()
