cmake_minimum_required(VERSION 3.5)
project(fic_prelogin LANGUAGES CXX)
include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/FicInstallLayout.cmake)
if(NOT TARGET fic-ipc)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../fic-common/fic-ipc ${CMAKE_CURRENT_BINARY_DIR}/fic-ipc)
endif()
if(NOT TARGET fic-core)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../fic-common/fic-core ${CMAKE_CURRENT_BINARY_DIR}/fic-core)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PRELOGIN_SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
add_library(fic-prelogin-core STATIC
    src/PreLoginController.cpp src/PreLoginStatusProvider.cpp src/PreLoginSystemdIntegration.cpp)
target_include_directories(fic-prelogin-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(fic-prelogin-core PUBLIC fic-ipc fic-core PkgConfig::PRELOGIN_SYSTEMD)
add_executable(fic-prelogin src/main.cpp)
target_link_libraries(fic-prelogin PRIVATE fic-prelogin-core)
install(TARGETS fic-prelogin RUNTIME DESTINATION "${FIC_PRIVATE_BINDIR}" COMPONENT fic-prelogin)
configure_file(resources/fic-prelogin.service.in fic-prelogin.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fic-prelogin.service
    DESTINATION "${FIC_SYSTEMD_UNIT_DIR}" COMPONENT fic-prelogin)
# Integration is activated by the checked lifecycle helper, never by copying
# a vendor DM unit or changing its alias link.
install(FILES resources/50-fic-prelogin.conf DESTINATION "${FIC_SHARE_DIR}/prelogin" COMPONENT fic-prelogin)
configure_file(resources/PreLoginPaths.h.in generated/PreLoginPaths.h @ONLY)
target_include_directories(fic-prelogin-core PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_executable(fic-prelogin-integration src/integration.cpp)
target_link_libraries(fic-prelogin-integration PRIVATE fic-prelogin-core)
install(TARGETS fic-prelogin-integration RUNTIME DESTINATION "${FIC_PRIVATE_BINDIR}" COMPONENT fic-prelogin)

option(FIC_PRELOGIN_GRAPHICS "Build the optional system Qt EGLFS/KMS frontend" ON)
if(FIC_PRELOGIN_GRAPHICS)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    get_target_property(FIC_PRELOGIN_QMAKE Qt6::qmake IMPORTED_LOCATION)
    execute_process(COMMAND "${FIC_PRELOGIN_QMAKE}" -query QT_INSTALL_PLUGINS
        OUTPUT_VARIABLE FIC_PRELOGIN_QT_PLUGIN_DIR OUTPUT_STRIP_TRAILING_WHITESPACE
        RESULT_VARIABLE FIC_PRELOGIN_QMAKE_RESULT)
    if(NOT FIC_PRELOGIN_QMAKE_RESULT EQUAL 0 OR NOT IS_ABSOLUTE "${FIC_PRELOGIN_QT_PLUGIN_DIR}")
        message(FATAL_ERROR "Cannot determine package-owned system Qt plugin directory")
    endif()
    set(FIC_PRELOGIN_FRONTEND_BINDIR "/usr/libexec/fic" CACHE PATH "Public unprivileged prelogin frontend directory")
    set(FIC_PRELOGIN_ALT_XGRP false)
    if(FIC_TARGET_PLATFORM STREQUAL "alt-p11")
        set(FIC_PRELOGIN_ALT_XGRP true)
    endif()
    configure_file(resources/PreLoginGraphicsPaths.h.in generated/PreLoginGraphicsPaths.h @ONLY)
    target_sources(fic-prelogin-core PRIVATE src/FrontendProcess.cpp)
    target_compile_definitions(fic-prelogin PRIVATE FIC_PRELOGIN_GRAPHICS=1)
    add_library(fic-prelogin-view STATIC src/QtPreLoginView.cpp)
    target_include_directories(fic-prelogin-view PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
    target_link_libraries(fic-prelogin-view PUBLIC Qt6::Widgets)
    add_executable(fic-prelogin-frontend src/frontend.cpp)
    target_include_directories(fic-prelogin-frontend PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(fic-prelogin-frontend PRIVATE fic-prelogin-view PkgConfig::PRELOGIN_SYSTEMD)
    install(TARGETS fic-prelogin-frontend RUNTIME DESTINATION "${FIC_PRELOGIN_FRONTEND_BINDIR}" COMPONENT fic-prelogin)
endif()
